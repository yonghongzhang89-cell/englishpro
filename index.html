list_frame.pack(pady=2, fill=tk.BOTH, padx=20, expand=True) self.tree = ttk.Treeview(list_frame, columns=("name", "size", "path"), show="headings", selectmode="extended", height=6) self.tree.heading("name", text="文件名"); self.tree.heading("size", text="大小"); self.tree.heading("path", text="完整路径") self.tree.column("name", width=300); self.tree.column("size", width=100, anchor="center"); self.tree.column("path", width=500) sc_y = ttk.Scrollbar(list_frame, orient=tk.VERTICAL, command=self.tree.yview); self.tree.configure(yscroll=sc_y.set) self.tree.pack(side=tk.LEFT, fill=tk.BOTH, expand=True); sc_y.pack(side=tk.RIGHT, fill=tk.Y) self.counter_label = tk.Label(self.root, text="就绪：待处理 0 个视频", font=("微软雅黑", 10)); self.counter_label.pack(pady=2) # 3. 处理选项 settings_frame = tk.LabelFrame(self.root, text=" 手机/循环优化选项 "); settings_frame.pack(pady=5, fill=tk.X, padx=20) tk.Label(settings_frame, text="格式:").grid(row=0, column=0, padx=10, pady=10) self.video_ext = ttk.Combobox(settings_frame, values=[".mp4"], width=8, state="readonly"); self.video_ext.set(".mp4"); self.video_ext.grid(row=0, column=1) tk.Label(settings_frame, text="字幕:").grid(row=0, column=2, padx=20) self.text_ext = ttk.Combobox(settings_frame, values=["SRT (推荐)"], width=12, state="readonly"); self.text_ext.set("SRT (推荐)"); self.text_ext.grid(row=0, column=3) # 4. 开始按钮 self.btn_start = tk.Button(self.root, text="🔥 开始生成全循/常亮手机视频", command=self.start_thread, bg="#4CAF50", fg="white", height=2, font=("微软雅黑", 12, "bold"), state=tk.DISABLED) self.btn_start.pack(pady=8, fill=tk.X, padx=20) self.timer_label = tk.Label(self.root, text="运行时间: 00:00:00", font=("Consolas", 11)); self.timer_label.pack() self.status_label = tk.Label(self.root, text="准备就绪", fg="blue"); self.status_label.pack(pady=2) self.progress = ttk.Progressbar(self.root, mode='determinate', maximum=100); self.progress.pack(pady=2, fill=tk.X, padx=20) # 底部红字 self.mode_label = tk.Label(self.root, text="工作模式：采用标准流写入，确保手机播放器单曲循环不中断、屏保不触发。", fg="#d32f2f", font=("微软雅黑", 9, "bold")) self.mode_label.pack(pady=5) def start_thread(self): if not self.all_files_data: return self.is_running = True; self.start_time = time.time(); self.update_timer() self.btn_start.config(state=tk.DISABLED); threading.Thread(target=self.process_logic, daemon=True).start() def process_logic(self): try: paths = [item['path'] for item in self.all_files_data] out_dir = os.path.dirname(paths[0]); target_v = os.path.join(out_dir, f"全循学习版_{int(time.time())}.mp4") if not self.model: self.model = whisper.load_model("base") if len(paths) > 1: self.set_progress(20, "🎬 正在极速合并 (标准流)...") clips = [VideoFileClip(f) for f in paths] # 获取原始帧率，这是同步的关键 fps = clips[0].fps or 24 # --- 核心回归：取消 resize 和复杂的 method，使用最稳的链式合并 --- final = concatenate_videoclips(clips) # 使用最基础的参数，保证手机系统能完美读取文件尾部 final.write_videofile(target_v, codec="libx264", audio_codec="aac", fps=fps, preset="ultrafast", logger=None) for c in clips: c.close() final.close() else: target_v = paths[0] self.set_progress(60, "✍️ 正在生成同步字幕...") res = self.model.transcribe(target_v, temperature=0, no_speech_threshold=0.4, condition_on_previous_text=False, fp16=False) srt_path = target_v.rsplit('.', 1)[0] + ".srt" with open(srt_path, "w", encoding="utf-8") as f: for idx, s in enumerate(res['segments']): if not s['text'].strip(): continue f.write(f"{idx+1}\n{self.fmt_t(s['start'])} --> {self.fmt_t(s['end'])}\n{s['text'].strip()}\n\n") self.set_progress(100, "🎉 处理完成") self.root.after(0, self.clear_list); messagebox.showinfo("成功", "已生成手机完美全循版视频！") except Exception as e: messagebox.showerror("错误", str(e)) finally: self.is_running = False; self.root.after(0, lambda: self.btn_start.config(state=tk.NORMAL)) # --- 通用逻辑保持不变 --- def fix_environment(self): try: ff_exe = resource_path("ffmpeg.exe") if os.path.exists(ff_exe): os.environ["IMAGEIO_FFMPEG_EXE"] = ff_exe bin_dir = os.path.dirname(ff_exe) if bin_dir not in os.environ["PATH"]: os.environ["PATH"] = bin_dir + os.pathsep + os.environ["PATH"] self.status_label.config(text="✅ 手机全循格式已就绪。", fg="#2e7d32") self.btn_start.config(state=tk.NORMAL) self.root.drop_target_register(DND_FILES); self.root.dnd_bind('<>', self.handle_drop) except Exception as e: self.status_label.config(text=f"⚠️ 环境警告: {e}", fg="red") def load_config(self): try: if os.path.exists(self.config_file): with open(self.config_file, "r", encoding="utf-8") as f: cfg = json.load(f); self.root.geometry(cfg.get("geometry", "1000x720")) cols = cfg.get("column_widths", {}) for col, width in cols.items(): self.tree.column(col, width=width) else: self.root.geometry("1000x720") except: self.root.geometry("1000x720") def on_closing(self): try: cfg = {"geometry": self.root.geometry(), "column_widths": {"name": self.tree.column("name", "width"), "size": self.tree.column("size", "width"), "path": self.tree.column("path", "width")}} with open(self.config_file, "w", encoding="utf-8") as f: json.dump(cfg, f, indent=4, ensure_ascii=False) except: pass self.root.destroy() def set_progress(self, val, text): self.root.after(0, lambda: self.progress.configure(value=val)); self.root.after(0, lambda: self.status_label.config(text=text)) def update_timer(self): if self.is_running: elapsed = int(time.time() - self.start_time); self.timer_label.config(text=f"运行时间: {str(datetime.timedelta(seconds=elapsed))}") self.root.after(1000, self.update_timer) def fmt_t(self, secs): h, m, s = int(secs//3600), int((secs%3600)//60), int(secs%60); ms = int((secs%1)*1000) return f"{h:02d}:{m:02d}:{s:02d},{ms:03d}" def add_files(self): self.process_files(filedialog.askopenfilenames(filetypes=[("视频", "*.mp4 *.mov *.avi *.mkv")])) def process_files(self, files): for f in files: f = f.strip().replace('"', '').replace("'", "") if os.path.exists(f): sz = os.path.getsize(f); self.all_files_data.append({'path': f, 'name': os.path.basename(f), 'size': f"{sz/(1024*1024):.2f} MB"}) self.refresh_tree() def refresh_tree(self): for i in self.tree.get_children(): self.tree.delete(i) for item in self.all_files_data: self.tree.insert("", tk.END, values=(item['name'], item['size'], item['path'])) def handle_drop(self, event): self.process_files(re.findall(r'\{(.*?)\}', event.data) if '{' in event.data else event.data.split()) def preview_selected(self): sel = self.tree.selection(); if sel: os.startfile(self.all_files_data[self.tree.index(sel[0])]['path']) def remove_selected(self): for i in sorted([self.tree.index(x) for x in self.tree.selection()], reverse=True): del self.all_files_data[i] self.refresh_tree() def clear_list(self): self.all_files_data = []; self.refresh_tree() def manual_dedup(self): seen = {}; to_keep = []; del_cnt = 0 for item in self.all_files_data: sz = os.path.getsize(item['path']) if sz in seen: try: os.remove(item['path']); del_cnt += 1 except: pass else: seen[sz] = True; to_keep.append(item) self.all_files_data = to_keep; self.refresh_tree(); messagebox.showinfo("去重", f"已清理 {del_cnt} 个文件。") if __name__ == "__main__": try: from tkinterdnd2 import TkinterDnD; root = TkinterDnD.Tk() except: root = tk.Tk() app = VideoToolApp(root); root.mainloop()
